# Configuração do NEAT (NeuroEvolution of Augmenting Topologies)
# Cada linha controla como as redes nascem, sofrem mutação e se reproduzem.
# Experimente mudar os valores e observe o efeito no treino!

[NEAT]
fitness_criterion     = max
# Meta de pontuação (só encerra o treino se no_fitness_termination = False)
fitness_threshold     = 5000
# Quantos dinos por geração
pop_size              = 50
reset_on_extinction   = False
# True = roda sempre o número de gerações pedido, mesmo se alguém bater a meta.
# Assim dá para ver a POPULAÇÃO inteira melhorando, não só o primeiro campeão.
no_fitness_termination = True

[DefaultGenome]
# --- funções de ativação dos neurônios ---
activation_default      = tanh
activation_mutate_rate  = 0.0
activation_options      = tanh

aggregation_default     = sum
aggregation_mutate_rate = 0.0
aggregation_options     = sum

# --- bias (viés) de cada neurônio ---
bias_init_mean          = 0.0
bias_init_stdev         = 1.0
bias_max_value          = 30.0
bias_min_value          = -30.0
bias_mutate_power       = 0.5
bias_mutate_rate        = 0.7
bias_replace_rate       = 0.1

# --- quando duas redes são "parecidas" (mesma espécie) ---
compatibility_disjoint_coefficient = 1.0
compatibility_weight_coefficient   = 0.5

# --- mutações de estrutura: adicionar/remover conexões e neurônios ---
conn_add_prob           = 0.5
conn_delete_prob        = 0.5
node_add_prob           = 0.2
node_delete_prob        = 0.2

enabled_default         = True
enabled_mutate_rate     = 0.01

feed_forward            = True
# Começa com todas as entradas ligadas a todas as saídas
initial_connection      = full

# --- formato da rede: 6 sensores -> 3 ações (nada, pular, abaixar) ---
num_hidden              = 0
num_inputs              = 6
num_outputs             = 3

response_init_mean      = 1.0
response_init_stdev     = 0.0
response_max_value      = 30.0
response_min_value      = -30.0
response_mutate_power   = 0.0
response_mutate_rate    = 0.0
response_replace_rate   = 0.0

# --- pesos das conexões (é aqui que o "aprendizado" acontece) ---
weight_init_mean        = 0.0
weight_init_stdev       = 1.0
weight_max_value        = 30
weight_min_value        = -30
weight_mutate_power     = 0.5
weight_mutate_rate      = 0.8
weight_replace_rate     = 0.1

[DefaultSpeciesSet]
compatibility_threshold = 3.0

[DefaultStagnation]
species_fitness_func = max
max_stagnation       = 20
species_elitism      = 2

[DefaultReproduction]
# Os 2 melhores passam intactos para a próxima geração
elitism            = 2
# Só os 20% melhores de cada espécie têm "filhos"
survival_threshold = 0.2
