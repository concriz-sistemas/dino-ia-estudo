# NEAT configuration (NeuroEvolution of Augmenting Topologies)
# Each line controls how the networks are born, mutate and reproduce.
# Try changing the values and watch the effect on training!

[NEAT]
fitness_criterion     = max
# Target score (only ends training if no_fitness_termination = False)
fitness_threshold     = 5000
# How many dinos per generation
pop_size              = 50
reset_on_extinction   = False
# True = always runs the requested number of generations, even if someone hits the target.
# That way you can see the whole POPULATION improving, not just the first champion.
no_fitness_termination = True

[DefaultGenome]
# --- the neurons' activation functions ---
activation_default      = tanh
activation_mutate_rate  = 0.0
activation_options      = tanh

aggregation_default     = sum
aggregation_mutate_rate = 0.0
aggregation_options     = sum

# --- each neuron's bias ---
bias_init_mean          = 0.0
bias_init_stdev         = 1.0
bias_max_value          = 30.0
bias_min_value          = -30.0
bias_mutate_power       = 0.5
bias_mutate_rate        = 0.7
bias_replace_rate       = 0.1

# --- when two networks are "similar" (same species) ---
compatibility_disjoint_coefficient = 1.0
compatibility_weight_coefficient   = 0.5

# --- structural mutations: adding/removing connections and neurons ---
conn_add_prob           = 0.5
conn_delete_prob        = 0.5
node_add_prob           = 0.2
node_delete_prob        = 0.2

enabled_default         = True
enabled_mutate_rate     = 0.01

feed_forward            = True
# Starts with every input connected to every output
initial_connection      = full

# --- network shape: 6 sensors -> 3 actions (nothing, jump, duck) ---
num_hidden              = 0
num_inputs              = 6
num_outputs             = 3

response_init_mean      = 1.0
response_init_stdev     = 0.0
response_max_value      = 30.0
response_min_value      = -30.0
response_mutate_power   = 0.0
response_mutate_rate    = 0.0
response_replace_rate   = 0.0

# --- connection weights (this is where the "learning" happens) ---
weight_init_mean        = 0.0
weight_init_stdev       = 1.0
weight_max_value        = 30
weight_min_value        = -30
weight_mutate_power     = 0.5
weight_mutate_rate      = 0.8
weight_replace_rate     = 0.1

[DefaultSpeciesSet]
compatibility_threshold = 3.0

[DefaultStagnation]
species_fitness_func = max
max_stagnation       = 20
species_elitism      = 2

[DefaultReproduction]
# The 2 best pass unchanged to the next generation
elitism            = 2
# Only the top 20% of each species have "children"
survival_threshold = 0.2
